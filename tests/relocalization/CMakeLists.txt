cmake_minimum_required(VERSION 3.10)
project(elevator_initial_pose_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
add_executable(initial_pose_test initial_pose_test.cpp)
target_include_directories(initial_pose_test PRIVATE ../../include)
target_link_libraries(initial_pose_test PRIVATE Eigen3::Eigen)
enable_testing()
add_test(NAME initial_pose_conversion COMMAND initial_pose_test)
